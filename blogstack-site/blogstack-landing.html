import { BlogStack } from '@blogstack/sdk'

// Initialize with your project key
const client = new BlogStack({ apiKey: 'bsk_live_...' })

// Fetch posts with full type safety
const { posts, total } = await client.posts.list({
  status: 'published',
  tag: 'engineering',
  limit: 10,
  page: 1,
})

// Fetch a single post by slug
const post = await client.posts.getBySlug('hello-world')

// Subscribe to new posts via webhook
client.webhooks.on('post.published', async (post) => {
  await revalidatePath(`/blog/${post.slug}`)
})

// SDK

Fetch posts
from anywhere

Works with every framework. TypeScript-first. Tiny bundle.

1

Install the SDK

npm i @blogstack/sdk — 3.2kb gzipped, zero dependencies.

2

Grab your project key

Create a project in the dashboard and copy your bsk_live_ key.

3

Fetch posts with full types

Use the React Query hooks or call the client directly in any server or edge context.

useBlog.ts 
// React Query hook (auto-generated)
import { usePosts } from '@blogstack/sdk/react'

function BlogPage() {
  const { data, isLoading } = usePosts({
    tag: 'engineering',
    limit: 6,
  })

  return data.posts.map(post => (
    <BlogCard
      key={post.id}
      post={post} // fully typed
    />
  ))
}

// Works with Next.js generateStaticParams
export async function generateStaticParams() {
  const posts = await client.posts.slugs()
  return posts.map(slug => ({ slug }))
}
